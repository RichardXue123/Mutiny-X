02. 武器与爆炸系统 (Weapons & Explosions)

全 15 种原版战术武器、工厂生成、独立接触动力学与 1:1 径向衰减爆炸冲量

1. 核心架构：武器基类与工厂装配

武器系统紧密依托于上一模块的离散物理系统。每种武器本质上都是一个具备独立触发规则的物理实体。系统采用了组合物理刚体 + 抽象模板方法模式 + 静态工厂模式：

状态生命周期： SpawnWeapon (创建并挂载) → Initialize(owner) (绑定物理碰撞盒与事件) → Twang / Fire (赋予初速度并通知回合状态机) → OnContact / Update (触发特化效果或引信) → Explosion.Spawn (产生范围爆炸) → Finish() (销毁并通知回合进入等待结算)。 
public abstract class MutinyWeapon : MonoBehaviour
{
    public string WeaponType = "weapon";
    public MutinyCharacter Owner;
    public bool IsFired { get; protected set; }
    public bool IsFinished { get; protected set; }

    // 组合自研 25 Hz 物理模拟组件
    public MutinyPhysicsBody PhysicsBody { get; protected set; }

    public virtual void Fire(Vector2 velocityPx)
    {
        IsFired = true;
        // 截断至武器最大初速 (默认 20 px/tick，香蕉/空投等为 30)
        if (velocityPx.sqrMagnitude > TwangMaxForce * TwangMaxForce)
            velocityPx = velocityPx.normalized * TwangMaxForce;

        PhysicsBody.SetVelocity(velocityPx.x, velocityPx.y);
        // 扣减行动权并通知 TurnManager 进入 ActionExecuting 阶段
        Owner.CanThrow = false;
        Owner.CanShoot = false;
        TurnManager.NotifyActionStarted();
    }

    // 子类重写接触反应：触碰墙体、触地、入水
    protected virtual void OnContact(CollisionSide side) { }
    protected virtual void Finish() { IsFinished = true; Destroy(gameObject); }
}

3. 统一爆炸计算核心：MutinyExplosion

爆炸不仅产生范围扣血，更是推动角色产生位移甚至将其击飞入海溺亡的关键机制。本项目 1:1 实现了原版 Explosion.as 严格公式：

爆炸核心计算公式

1. 作用范围计算：

Radius = (Size * 0.5f) + 20.0f;

例如：CherryBomb (Size=80) 对应半径 60px；Dynamite (Size=250) 对应半径 145px。

2. 线性距离衰减比率 (Ratio)：

ratio = 1.0f - (distance / Radius);

中心为 1.0，边缘衰减至 0.0。

3. 伤害结算：

Damage = MaxDamage * ratio;

4. 向上抛弹冲击力 (Upward Pop Impulse)：

force = 0.06f * ratio * MaxDamage;

vx += normX * 5.0f * force;

vy += (normY * 5.0f * force) - (force * 6.0f);

注：Flash 像素坐标系中 -Y 为向上，因此减去 force * 6 会给予角色显著的腾空跳跃冲量。

帧驱动生命周期：第 3 帧生效机制

private void Update()
{
    m_FrameTimer += Time.deltaTime;
    float frameDuration = 1f / FrameRate; // 25 Hz Nitrome 基准

    while (m_FrameTimer >= frameDuration)
    {
        m_FrameTimer -= frameDuration;
        m_CurrentFrame++;

        // 原版 AS2 字节码：伤害判定必须在第 3 帧 (0-indexed 的 frame 2) 触发
        if (m_CurrentFrame == 2 && !m_HitApplied)
        {
            ApplyHit();
            m_HitApplied = true;
        }

        // 第 8 帧动画播放完毕，自我销毁
        if (m_CurrentFrame >= 8)
        {
            Destroy(gameObject);
            return;
        }
    }
}

4. 代表性武器实现细节对照

CherryBomb（触碰即爆）

注册在 PhysicsBody 上的碰撞回调：触碰任意地板、墙体或落水，直接原地引爆并触发音效：

protected override void OnContact(CollisionSide side)
{
    if (IsFinished) return;
    MutinyExplosion.Spawn(
        new Vector2(PhysicsBody.State.X, PhysicsBody.State.Y),
        size: 80f, maxDamage: 40f, owner: Owner);
    Finish();
}

Dynamite（静止引爆 / 落水熄火）

重写物理特性，触碰不爆炸，落水熄灭引信，唯有在地面完全静止时引爆：

private void Update()
{
    if (!IsFired || IsFinished) return;
    // 落水熄灭
    if (PhysicsBody.IsInWater)
    {
        // 切换为熄灭贴图，不产生爆炸
        Finish();
        return;
    }
    // 仅在静止时巨爆
    if (PhysicsBody.IsAtRest)
    {
        MutinyExplosion.Spawn(pos, size: 250f, maxDamage: 70f);
        Finish();
    }
}
